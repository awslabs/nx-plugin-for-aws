import Infrastructure from '@components/infrastructure.astro';

Để triển khai MCP Server của bạn, các tệp sau được tạo ra:

<Infrastructure>
<Fragment slot="cdk">
<FileTree>
  - packages/common/constructs/src
    - app
      - mcp-servers
        - \<mcp-server-name>
          - \<mcp-server-name>.ts CDK construct for deploying your MCP Server
</FileTree>
</Fragment>
<Fragment slot="terraform">
<FileTree>
  - packages/common/terraform/src
    - app
      - mcp-servers
        - \<mcp-server-name>
          - \<mcp-server-name>.tf Module for deploying your MCP Server
    - core
      - agent-core
        - runtime.tf Generic module for deploying to Bedrock AgentCore Runtime
      - agent-core-code (when `infra` is `agentcore`)
        - runtime.tf Packages your MCP Server's code and delegates to `agent-core`
      - agent-core-container (when `infra` is `agentcore-ecr`)
        - runtime.tf Builds and publishes your MCP Server's image and delegates to `agent-core`
</FileTree>
</Fragment>
</Infrastructure>
